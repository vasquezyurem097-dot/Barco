#include <stdio.h>

#define FILAS 6
#define COLUMNAS 7

// Inicializa el tablero vacio
void inicializarTablero(char tablero[FILAS][COLUMNAS])
{
    int i, j;

    for(i = 0; i < FILAS; i++)
    {
        for(j = 0; j < COLUMNAS; j++)
        {
            tablero[i][j] = '.';
        }
    }
}

// Muestra el tablero
void mostrarTablero(char tablero[FILAS][COLUMNAS])
{
    int i, j;

    printf("\n");

    for(i = 0; i < FILAS; i++)
    {
        printf("|");

        for(j = 0; j < COLUMNAS; j++)
        {
            printf(" %c |", tablero[i][j]);
        }

        printf("\n");
    }

    printf("  1   2   3   4   5   6   7\n\n");
}

// Coloca la ficha en la columna elegida
int colocarFicha(char tablero[FILAS][COLUMNAS], int columna, char ficha)
{
    int i;

    for(i = FILAS - 1; i >= 0; i--)
    {
        if(tablero[i][columna] == '.')
        {
            tablero[i][columna] = ficha;
            return 1;
        }
    }

    return 0;
}

// Revisa si hay cuatro fichas seguidas
int revisarGanador(char tablero[FILAS][COLUMNAS], char ficha)
{
    int i, j;

    // Revisar horizontal
    for(i = 0; i < FILAS; i++)
    {
        for(j = 0; j < COLUMNAS - 3; j++)
        {
            if(tablero[i][j] == ficha &&
               tablero[i][j + 1] == ficha &&
               tablero[i][j + 2] == ficha &&
               tablero[i][j + 3] == ficha)
            {
                return 1;
            }
        }
    }

    // Revisar vertical
    for(i = 0; i < FILAS - 3; i++)
    {
        for(j = 0; j < COLUMNAS; j++)
        {
            if(tablero[i][j] == ficha &&
               tablero[i + 1][j] == ficha &&
               tablero[i + 2][j] == ficha &&
               tablero[i + 3][j] == ficha)
            {
                return 1;
            }
        }
    }

    // Revisar diagonal hacia abajo
    for(i = 0; i < FILAS - 3; i++)
    {
        for(j = 0; j < COLUMNAS - 3; j++)
        {
            if(tablero[i][j] == ficha &&
               tablero[i + 1][j + 1] == ficha &&
               tablero[i + 2][j + 2] == ficha &&
               tablero[i + 3][j + 3] == ficha)
            {
                return 1;
            }
        }
    }

    // Revisar diagonal hacia arriba
    for(i = 3; i < FILAS; i++)
    {
        for(j = 0; j < COLUMNAS - 3; j++)
        {
            if(tablero[i][j] == ficha &&
               tablero[i - 1][j + 1] == ficha &&
               tablero[i - 2][j + 2] == ficha &&
               tablero[i - 3][j + 3] == ficha)
            {
                return 1;
            }
        }
    }

    return 0;
}

// Revisa si el tablero esta lleno
int tableroLleno(char tablero[FILAS][COLUMNAS])
{
    int j;

    for(j = 0; j < COLUMNAS; j++)
    {
        if(tablero[0][j] == '.')
        {
            return 0;
        }
    }

    return 1;
}

int main()
{
    char tablero[FILAS][COLUMNAS];
    int columna;
    int jugador = 1;
    int terminado = 0;
    char ficha;

    inicializarTablero(tablero);

    printf("===== CONECTA 4 =====\n");
    printf("Jugador 1: X\n");
    printf("Jugador 2: O\n");
    printf("Conecta cuatro fichas para ganar.\n");

    while(terminado == 0)
    {
        mostrarTablero(tablero);

        if(jugador == 1)
        {
            ficha = 'X';
            printf("Turno del Jugador 1 (X)\n");
        }
        else
        {
            ficha = 'O';
            printf("Turno del Jugador 2 (O)\n");
        }

        printf("Elige una columna del 1 al 7: ");

        if(scanf("%d", &columna) != 1)
        {
            printf("Entrada invalida. Fin del juego.\n");
            return 1;
        }

        columna = columna - 1;

        if(columna < 0 || columna >= COLUMNAS)
        {
            printf("Columna invalida. Intenta de nuevo.\n");
        }
        else if(colocarFicha(tablero, columna, ficha) == 0)
        {
            printf("Esa columna esta llena. Elige otra.\n");
        }
        else
        {
            if(revisarGanador(tablero, ficha) == 1)
            {
                mostrarTablero(tablero);
                printf("¡Gano el Jugador %d!\n", jugador);
                terminado = 1;
            }
            else if(tableroLleno(tablero) == 1)
            {
                mostrarTablero(tablero);
                printf("¡Empate! El tablero esta lleno.\n");
                terminado = 1;
            }
            else
            {
                if(jugador == 1)
                {
                    jugador = 2;
                }
                else
                {
                    jugador = 1;
                }
            }
        }
    }

    printf("===== FIN DEL JUEGO =====\n");

    return 0;
}
